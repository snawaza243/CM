CREATE TABLE temp_client_master (
    ClientID NUMBER PRIMARY KEY,
    ClientCode VARCHAR2(20) UNIQUE,          -- TC + ID
    FirstName VARCHAR2(100) NOT NULL,
    LastName VARCHAR2(100) NOT NULL,
    Email VARCHAR2(255) NOT NULL,
    Phone VARCHAR2(20),
    Address VARCHAR2(500),
    PasswordHash VARCHAR2(500),
    CreatedBy VARCHAR2(100),
    CreatedDate DATE DEFAULT SYSDATE,
    ModifiedBy VARCHAR2(100),
    ModifiedDate DATE,
    LastLoginDateTime DATE
);


-- Sequence starts from 1, increment by 1
CREATE SEQUENCE Seq_Temp_Client_ID
    START WITH 1
    INCREMENT BY 1;


CREATE OR REPLACE TRIGGER trg_SetClientCode
BEFORE INSERT ON temp_client_master
FOR EACH ROW
BEGIN
    -- Assign ClientID from sequence if not provided
    IF :NEW.ClientID IS NULL THEN
        SELECT Seq_Temp_Client_ID.NEXTVAL
        INTO :NEW.ClientID
        FROM dual;
    END IF;

    -- Generate ClientCode as 'TC' + zero-padded ClientID (e.g., TC0001)
    :NEW.ClientCode := 'TC' || LPAD(:NEW.ClientID, 4, '0');

    -- Set CreatedDate if not provided
    IF :NEW.CreatedDate IS NULL THEN
        :NEW.CreatedDate := SYSDATE;
    END IF;
END;
/
