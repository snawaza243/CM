CREATE OR REPLACE PROCEDURE PROC_CM_SERVICE_SAVE (
    p_service_code       IN OUT VARCHAR2,    -- SRV_001
    p_service_name       IN VARCHAR2,
    p_service_desc       IN CLOB,
    p_service_icon       IN VARCHAR2,        -- New: Icon name (fa-tools, fa-building, etc.)
    p_service_type       IN VARCHAR2,
    p_service_type_code  IN OUT VARCHAR2,    -- SRVT_001
    p_created_by         IN VARCHAR2,
    p_active             IN CHAR DEFAULT 'Y',
    p_status             OUT VARCHAR2,
    p_message            OUT VARCHAR2
)
AS
    v_count        NUMBER;
    v_next_srv_cd  NUMBER;
    v_next_srvt_cd NUMBER;
BEGIN
    SAVEPOINT sv_tr;

    -- ✅ Ensure SERVICE_TYPE_CODE always exists
    IF p_service_type_code IS NULL THEN
        SELECT NVL(MAX(TO_NUMBER(SUBSTR(SERVICE_TYPE_CODE,6))), 0)
          INTO v_next_srvt_cd
          FROM CM_LIST_SERVICES
         WHERE SERVICE_TYPE_CODE LIKE 'SRVT_%';

        v_next_srvt_cd := v_next_srvt_cd + 1;
        p_service_type_code := 'SRVT_' || LPAD(v_next_srvt_cd, 3, '0');
    END IF;

    -- ✅ Check if SERVICE_CODE exists (for update)
    IF p_service_code IS NOT NULL THEN
        SELECT COUNT(*) INTO v_count
        FROM CM_LIST_SERVICES
        WHERE SERVICE_CODE = p_service_code;

        IF v_count > 0 THEN
            UPDATE CM_LIST_SERVICES
               SET SERVICE_NAME      = p_service_name,
                   SERVICE_DESC      = p_service_desc,
                   SERVICE_ICON      = p_service_icon,
                   SERVICE_TYPE      = p_service_type,
                   SERVICE_TYPE_CODE = p_service_type_code,
                   ACTIVE            = p_active,
                   MODIFIED_BY       = p_created_by,
                   MODIFIED_ON       = SYSDATE
             WHERE SERVICE_CODE = p_service_code;

            p_status  := '1';
            p_message := 'Service updated successfully. CODE: ' || p_service_code;
            DBMS_OUTPUT.PUT_LINE(p_message);
            RETURN;
        END IF;
    END IF;

    -- ✅ Generate new SERVICE_CODE if inserting
    SELECT NVL(MAX(TO_NUMBER(SUBSTR(SERVICE_CODE,5))), 0)
      INTO v_next_srv_cd
      FROM CM_LIST_SERVICES
     WHERE SERVICE_CODE LIKE 'SRV_%';

    v_next_srv_cd := v_next_srv_cd + 1;
    p_service_code := 'SRV_' || LPAD(v_next_srv_cd, 3, '0');

    -- ✅ Insert new record
    INSERT INTO CM_LIST_SERVICES (
        SERVICE_CODE, SERVICE_NAME, SERVICE_DESC, SERVICE_ICON,
        SERVICE_TYPE, SERVICE_TYPE_CODE, CREATED_BY, CREATED_ON, ACTIVE
    ) VALUES (
        p_service_code, p_service_name, p_service_desc, p_service_icon,
        p_service_type, p_service_type_code, p_created_by, SYSDATE, p_active
    );

    p_status  := '1';
    p_message := 'Inserted new service successfully. CODE: ' || p_service_code;
    DBMS_OUTPUT.PUT_LINE(p_message);

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK TO sv_tr;
        p_status := '0';
        IF p_service_code IS NULL THEN
            p_message := 'Insertion failed: ' || SQLERRM;
        ELSE
            p_message := 'Update failed: ' || SQLERRM;
        END IF;
        DBMS_OUTPUT.PUT_LINE(p_message);
END;
