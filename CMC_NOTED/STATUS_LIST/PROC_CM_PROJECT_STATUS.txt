CREATE OR REPLACE PROCEDURE PROC_CM_PROJECT_STATUS(
    P_STATUS_ID       IN  NUMBER,
    P_STATUS_NAME     IN  VARCHAR2,
    P_STATUS_TYPE     IN  VARCHAR2,
    P_STATUS_ICON     IN  VARCHAR2,
    P_STATUS_BG_CLASS IN  VARCHAR2,
    P_STATUS_DESC     IN  VARCHAR2,
    P_STATUS_ACTIVE   IN  CHAR,
    P_CREATED_BY      IN  VARCHAR2,
    OUT_STATUS        OUT VARCHAR2,
    OUT_MESSAGE       OUT VARCHAR2
)
IS
    V_COUNT NUMBER;
BEGIN
    IF P_STATUS_ID IS NOT NULL THEN
        SELECT COUNT(*) INTO V_COUNT FROM CM_PROJECT_STATUS WHERE STATUS_ID = P_STATUS_ID;
        IF V_COUNT > 0 THEN
            UPDATE CM_PROJECT_STATUS
            SET STATUS_NAME     = P_STATUS_NAME,
                STATUS_TYPE     = P_STATUS_TYPE,
                STATUS_ICON     = P_STATUS_ICON,
                STATUS_BG_CLASS = P_STATUS_BG_CLASS,
                STATUS_DESC     = P_STATUS_DESC,
                STATUS_ACTIVE   = P_STATUS_ACTIVE,
                MODIFY_ON       = SYSDATE,
                MODIFY_BY       = P_CREATED_BY
            WHERE STATUS_ID = P_STATUS_ID;

            OUT_STATUS := 'SUCCESS';
            OUT_MESSAGE := 'Record updated';
        ELSE
            OUT_STATUS := 'ERROR';
            OUT_MESSAGE := 'Record not found';
        END IF;
    ELSE
        INSERT INTO CM_PROJECT_STATUS(
            STATUS_NAME, STATUS_TYPE, STATUS_ICON, STATUS_BG_CLASS, STATUS_DESC, STATUS_ACTIVE, CREATED_BY
        )
        VALUES (
            P_STATUS_NAME, P_STATUS_TYPE, P_STATUS_ICON, P_STATUS_BG_CLASS, P_STATUS_DESC, P_STATUS_ACTIVE, P_CREATED_BY
        );

        OUT_STATUS := 'SUCCESS';
        OUT_MESSAGE := 'Record inserted';
    END IF;

    COMMIT;
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        OUT_STATUS := 'ERROR';
        OUT_MESSAGE := SQLERRM;
END;
/
