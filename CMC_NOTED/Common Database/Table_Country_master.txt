CREATE TABLE COUNTRY_MASTER (
    COUNTRY_ID       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    COUNTRY_NAME     VARCHAR2(100) NOT NULL,
    CREATED_BY       VARCHAR2(50) DEFAULT NULL,
    CREATED_DATE     DATE DEFAULT NULL,
    UPDATED_BY       VARCHAR2(50) DEFAULT NULL,
    UPDATED_DATE     DATE DEFAULT NULL,
    STATE_COUNT      NUMBER DEFAULT NULL,
    COUNTRY_CODE     VARCHAR2(10) DEFAULT NULL,
    CITY_COUNT       NUMBER DEFAULT NULL,
    CURRENCY_CODE    VARCHAR2(10) DEFAULT NULL,
    CURRENCY_NAME    VARCHAR2(50) DEFAULT NULL,
    TIME_ZONE        VARCHAR2(50) DEFAULT NULL,
    DIALING_CODE     VARCHAR2(10) DEFAULT NULL,
    CONTINENT_NAME   VARCHAR2(50) DEFAULT NULL,
    TIMEST           TIMESTAMP DEFAULT NULL
);
