CREATE TABLE CM_PROJECT_STATUS
(
    STATUS_ID       NUMBER PRIMARY KEY,
    STATUS_NAME     VARCHAR2(100) NOT NULL,
    STATUS_TYPE     VARCHAR2(50),
    STATUS_ICON     VARCHAR2(50),
    STATUS_BG_CLASS VARCHAR2(50),
    STATUS_DESC     VARCHAR2(4000),
    STATUS_ACTIVE   CHAR(1) DEFAULT 'Y',
    CREATE_ON       DATE DEFAULT SYSDATE,
    CREATED_BY      VARCHAR2(50),
    MODIFY_ON       DATE,
    MODIFY_BY       VARCHAR2(50)
);
