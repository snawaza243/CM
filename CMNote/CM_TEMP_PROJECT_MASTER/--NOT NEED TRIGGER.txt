
CREATE OR REPLACE TRIGGER TRG_CM_TEMP_PROJECT_MASTER_BI
BEFORE INSERT ON CM_TEMP_PROJECT_MASTER
FOR EACH ROW
DECLARE
  v_next_id NUMBER;
BEGIN
  IF :NEW.PROJECT_ID IS NULL THEN
    SELECT SEQ_CM_TEMP_PROJECT_MASTER.NEXTVAL INTO v_next_id FROM DUAL;
    :NEW.PROJECT_ID := v_next_id;
  ELSE
    v_next_id := :NEW.PROJECT_ID;
  END IF;

  :NEW.PROJECT_CODE := 'TP_' || TO_CHAR(v_next_id);

  IF :NEW.CREATED_AT IS NULL THEN
    :NEW.CREATED_AT := SYSDATE;
  END IF;
END;
/
