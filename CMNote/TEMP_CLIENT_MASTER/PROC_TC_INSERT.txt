create or replace NONEDITIONABLE PROCEDURE PROC_TC_INSERT(
    P_FIRSTNAME      IN  VARCHAR2,
    P_LASTNAME       IN  VARCHAR2,
    P_EMAIL          IN  VARCHAR2,
    P_PHONE          IN  VARCHAR2 DEFAULT NULL,
    P_ADDRESS        IN  VARCHAR2 DEFAULT NULL,
    P_PASSWORDHASH   IN  VARCHAR2 DEFAULT NULL,
    P_CREATEDBY      IN  VARCHAR2 DEFAULT NULL,
    p_ProfileImage   IN BLOB,
    P_CLIENTCODE     OUT VARCHAR2,
    P_SUCCESS        OUT NUMBER,
    P_MESSAGE        OUT VARCHAR2
)
AS
    V_SAVEPOINT_NAME VARCHAR2(30);
BEGIN
    -- Create savepoint for this operation
    V_SAVEPOINT_NAME := 'sp_insert_save';
    SAVEPOINT SP_INSERT_SAVE;

    BEGIN
        INSERT INTO TEMP_CLIENT_MASTER (
            FIRSTNAME,
            LASTNAME,
            EMAIL,
            PHONE,
            ADDRESS,
            PASSWORDHASH,
            CREATEDBY,
            PROFILE_IMAGE
        ) VALUES (
            P_FIRSTNAME,
            P_LASTNAME,
            P_EMAIL,
            P_PHONE,
            P_ADDRESS,
            P_PASSWORDHASH,
            P_CREATEDBY,
            p_ProfileImage
        )
        RETURNING CLIENTCODE INTO P_CLIENTCODE;

        P_SUCCESS := 1;
        P_MESSAGE := 'Client created successfully.';
        COMMIT; -- Commit only if successful

    EXCEPTION
        WHEN DUP_VAL_ON_INDEX THEN
            ROLLBACK TO SP_INSERT_SAVE;
            P_SUCCESS := 0;
            P_MESSAGE := 'Error: Email or ClientCode already exists.';
        WHEN OTHERS THEN
            ROLLBACK TO SP_INSERT_SAVE;
            P_SUCCESS := 0;
            P_MESSAGE := 'Error: ' || SQLERRM;
    END;
END;
