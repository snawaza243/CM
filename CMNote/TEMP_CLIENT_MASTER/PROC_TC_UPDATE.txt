create or replace NONEDITIONABLE PROCEDURE PROC_TC_UPDATE(
    P_CLIENTCODE     IN  VARCHAR2,
    P_FIRSTNAME      IN  VARCHAR2 DEFAULT NULL,
    P_LASTNAME       IN  VARCHAR2 DEFAULT NULL,
    P_EMAIL          IN  VARCHAR2 DEFAULT NULL,
    P_PHONE          IN  VARCHAR2 DEFAULT NULL,
    P_ADDRESS        IN  VARCHAR2 DEFAULT NULL,
    P_PASSWORDHASH   IN  VARCHAR2 DEFAULT NULL,  -- Only update if not NULL
    P_MODIFIEDBY     IN  VARCHAR2 DEFAULT NULL,
    p_ProfileImage   IN BLOB,
    P_SUCCESS        OUT NUMBER,
    P_MESSAGE        OUT VARCHAR2
)
AS
    V_SAVEPOINT_NAME VARCHAR2(30);
BEGIN
    V_SAVEPOINT_NAME := 'sp_update_save';
    SAVEPOINT SP_UPDATE_SAVE;

    BEGIN
        UPDATE TEMP_CLIENT_MASTER
        SET
            FIRSTNAME = NVL(P_FIRSTNAME, FIRSTNAME),
            LASTNAME = NVL(P_LASTNAME, LASTNAME),
            EMAIL = NVL(P_EMAIL, EMAIL),
            PHONE = NVL(P_PHONE, PHONE),
            ADDRESS = NVL(P_ADDRESS, ADDRESS),
            PASSWORDHASH = CASE WHEN P_PASSWORDHASH IS NOT NULL THEN P_PASSWORDHASH ELSE PASSWORDHASH END,
            MODIFIEDBY = P_MODIFIEDBY,
            MODIFIEDDATE = SYSDATE,
            PROFILE_IMAGE = p_ProfileImage
        WHERE ClientCode = P_CLIENTCODE;

        IF SQL%ROWCOUNT = 0 THEN
            ROLLBACK TO SP_UPDATE_SAVE;
            P_SUCCESS := 0;
            P_MESSAGE := 'No client found with the given ClientID.';
        ELSE
            COMMIT;
            P_SUCCESS := 1;
            P_MESSAGE := 'Client updated successfully.';
        END IF;

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK TO SP_UPDATE_SAVE;
            P_SUCCESS := 0;
            P_MESSAGE := 'Error: ' || SQLERRM;
    END;
END;
