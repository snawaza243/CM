CREATE OR REPLACE PROCEDURE PROC_TC_GET_USER(
    P_CLIENTID    IN  NUMBER DEFAULT NULL,
    P_EMAIL       IN  VARCHAR2 DEFAULT NULL,
    P_PHONE       IN  VARCHAR2 DEFAULT NULL,
    P_CLIENTCODE  OUT VARCHAR2,
    P_PASSWORDHASH OUT VARCHAR2
)
AS
BEGIN
    SELECT CLIENTCODE, PASSWORDHASH
    INTO P_CLIENTCODE, P_PASSWORDHASH
    FROM (
        SELECT CLIENTCODE, PASSWORDHASH
        FROM TEMP_CLIENT_MASTER
        WHERE (P_CLIENTID IS NOT NULL AND CLIENTID = P_CLIENTID)
           OR (P_EMAIL IS NOT NULL AND LOWER(EMAIL) = LOWER(P_EMAIL))
           OR (P_PHONE IS NOT NULL AND PHONE = P_PHONE)
        ORDER BY CLIENTID
    )
    WHERE ROWNUM = 1;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        P_CLIENTCODE := NULL;
        P_PASSWORDHASH := NULL;
END;
