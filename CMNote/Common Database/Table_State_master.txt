CREATE TABLE WEALTHMAKER.STATE_MASTER (
    STATE_ID            NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    STATE_NAME          VARCHAR2(100) NOT NULL,
    STATE_CODE          VARCHAR2(10) NOT NULL,
    COUNTRY_ID          NUMBER NOT NULL,
    CAPITAL_NAME        VARCHAR2(100) DEFAULT NULL,
    CITY_COUNT          NUMBER DEFAULT 0,
    IS_UNION_TERRITORY  CHAR(1) DEFAULT 'N' CHECK (IS_UNION_TERRITORY IN ('Y', 'N')),
    TIME_ZONE           VARCHAR2(100) DEFAULT 'Asia/Kolkata',
    STATUS              CHAR(1) DEFAULT 'A' CHECK (STATUS IN ('A', 'I')),
    CREATED_BY          VARCHAR2(50) DEFAULT 'SYSTEM',
    CREATED_DATE        DATE DEFAULT SYSDATE,
    UPDATED_BY          VARCHAR2(50) DEFAULT NULL,
    UPDATED_DATE        DATE DEFAULT NULL,
    CONSTRAINT FK_STATE_COUNTRY FOREIGN KEY (COUNTRY_ID)
        REFERENCES WEALTHMAKER.COUNTRY_MASTER (COUNTRY_ID)
);
