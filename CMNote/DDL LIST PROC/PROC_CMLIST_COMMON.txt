create or replace NONEDITIONABLE PROCEDURE PROC_CMLIST_COMMON(
    P_FOR     IN  VARCHAR2,
    P_BY      IN  VARCHAR2,
    P_X       IN  VARCHAR2,
    P_ROLE    IN  VARCHAR2,
    P_LOGID   IN  VARCHAR2,
    P_SUCCESS  OUT NUMBER,
    P_MESSAGE OUT VARCHAR2,
    P_CURSOR  OUT SYS_REFCURSOR
) AS
BEGIN
    IF P_FOR = 'DDL_CM_SERVICES' THEN
        P_SUCCESS := 1;
        P_MESSAGE := 'VALID LIST EXISTS';

        OPEN P_CURSOR FOR
        SELECT SERVICE_CODE AS CODE,
               SERVICE_NAME AS NAME
        FROM CM_LIST_SERVICES
        WHERE ACTIVE = 'Y';
    END IF;
EXCEPTION
    WHEN OTHERS THEN
        P_SUCCESS := 0;
        P_MESSAGE := SQLERRM;
END PROC_CMLIST_COMMON;
