create or replace NONEDITIONABLE PROCEDURE PROC_CM_GET_DATA_COUNT(
    P_FOR        IN  VARCHAR2,
    P_BY         IN  VARCHAR2,
    P_BY_VALUE   IN  VARCHAR2,
    P_LOG_ID     IN  VARCHAR2,
    P_ROLE_ID    IN  VARCHAR2,
    P_SUCCESS    OUT VARCHAR2,       -- 0 OR 1 
    P_MESSAGE    OUT VARCHAR2,       -- MESSAGE IF ANY ERROR
    P_COUNT      OUT VARCHAR2,       -- RETURNING COUNT DATA AS PER NEEDED     
    P_CURSOR     OUT SYS_REFCURSOR   -- RETURNING DATA IN CURSOR AS PER NEEDED
)
IS
    V_COUNT  NUMBER := 0;
BEGIN
    IF P_FOR = 'CM_PROJECT_LIST_1' THEN
        SELECT COUNT(*)
        INTO V_COUNT
        FROM CM_TEMP_PROJECT_MASTER
        WHERE TRIM(APPROVED_STATUS) IS NULL
           OR SUBSTR(UPPER(APPROVED_STATUS), 1, 1) = 'P';

        IF V_COUNT > 0 THEN
            P_SUCCESS := '1';
            P_MESSAGE := 'PENDING DATA FOUND';
            P_COUNT   := V_COUNT;

            OPEN P_CURSOR FOR
                --SELECT * FROM CM_TEMP_PROJECT_MASTER 
                --WHERE TRIM(APPROVED_STATUS) IS NULL OR SUBSTR(UPPER(APPROVED_STATUS), 1, 1) = 'P';

                SELECT 
                -- COUT OF EARCH ROW
                PROJECT_ID,PROJECT_CODE, CLIENT_CODE, CLIENT_NAME, EMAIL, PHONE,OTHER_CONTACTS, PROJECT_TYPE, 
                PROJECT_ADDRESS, PROJECT_DESCRIPTION, ESTIMATED_BUDGET, TIMELINE, 
                FN_FORMAT_DATE(EXPECTED_START_DATE, 'DD/MM/YYYY') as EXPECTED_START_DATE , 
                FN_FORMAT_DATE(EXPECTED_DEADLINE, 'DD/MM/YYYY') as EXPECTED_DEADLINE , APPROVED_STATUS, APPROVED_BY, CURRENT_STATUS, 
                FN_FORMAT_DATE(CREATED_AT, 'DD/MM/YYYY') as CREATED_AT , 
                FN_FORMAT_DATE(UPDATED_AT, 'DD/MM/YYYY') as UPDATED_AT , 
                UPDATED_BY, PROJECT_NAME, REFERENCE_NAME
                FROM CM_TEMP_PROJECT_MASTER
                WHERE TRIM(APPROVED_STATUS) IS NULL
                OR SUBSTR(UPPER(APPROVED_STATUS), 1, 1) = 'P';

        ELSE
            P_SUCCESS := '0';
            P_MESSAGE := 'NO PENDING DATA';
            P_COUNT   := '0';
            P_CURSOR  := NULL;
        END IF;
        
    ELSIF P_FOR = 'CM_PROJECT_LIST_2' THEN
        SELECT COUNT(*)
        INTO V_COUNT
        FROM CM_TEMP_PROJECT_MASTER
        WHERE TRIM(APPROVED_STATUS) IS NULL
           OR SUBSTR(UPPER(APPROVED_STATUS), 1, 1) = 'A';

        IF V_COUNT > 0 THEN
            P_SUCCESS := '1';
            P_MESSAGE := 'APPROVED DATA FOUND';
            P_COUNT   := V_COUNT;

            OPEN P_CURSOR FOR
                --SELECT * FROM CM_TEMP_PROJECT_MASTER 
                --WHERE TRIM(APPROVED_STATUS) IS NULL OR SUBSTR(UPPER(APPROVED_STATUS), 1, 1) = 'P';

                SELECT 
                -- COUT OF EARCH ROW
                PROJECT_ID,PROJECT_CODE, CLIENT_CODE, CLIENT_NAME, EMAIL, PHONE,OTHER_CONTACTS, PROJECT_TYPE, 
                PROJECT_ADDRESS, PROJECT_DESCRIPTION, ESTIMATED_BUDGET, TIMELINE, 
                FN_FORMAT_DATE(EXPECTED_START_DATE, 'DD/MM/YYYY') as EXPECTED_START_DATE , 
                FN_FORMAT_DATE(EXPECTED_DEADLINE, 'DD/MM/YYYY') as EXPECTED_DEADLINE , APPROVED_STATUS, APPROVED_BY, CURRENT_STATUS, 
                FN_FORMAT_DATE(CREATED_AT, 'DD/MM/YYYY') as CREATED_AT , 
                FN_FORMAT_DATE(UPDATED_AT, 'DD/MM/YYYY') as UPDATED_AT , 
                UPDATED_BY, PROJECT_NAME, REFERENCE_NAME
                FROM CM_TEMP_PROJECT_MASTER
                WHERE TRIM(APPROVED_STATUS) IS NULL
                OR SUBSTR(UPPER(APPROVED_STATUS), 1, 1) = 'A';

        ELSE
            P_SUCCESS := '0';
            P_MESSAGE := 'NO APPROVED DATA';
            P_COUNT   := '0';
            P_CURSOR  := NULL;
        END IF;
    ELSE
        P_SUCCESS := '0';
        P_MESSAGE := 'INVALID P_FOR VALUE';
        P_COUNT   := '0';
        P_CURSOR  := NULL;
    END IF;

EXCEPTION
    WHEN OTHERS THEN
        P_SUCCESS := '0';
        P_MESSAGE := 'ERROR: ' || SQLERRM;
        P_COUNT   := '0';
        P_CURSOR  := NULL;
END PROC_CM_GET_DATA_COUNT;
