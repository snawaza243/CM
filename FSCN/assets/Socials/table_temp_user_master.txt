CREATE TABLE TEMP_USER_MASTER (
    USER_ID             NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FIRST_NAME          VARCHAR2(100) NOT NULL,
    LAST_NAME           VARCHAR2(100) NOT NULL,
    EMAIL               VARCHAR2(150) UNIQUE NOT NULL,
    PHONE               VARCHAR2(15),
    PASSWORD_HASH       VARCHAR2(255) NOT NULL,
    MARKETING_EMAILS    CHAR(1) DEFAULT 'N' CHECK (MARKETING_EMAILS IN ('Y', 'N')),
    TERMS_ACCEPTED      CHAR(1) DEFAULT 'Y' CHECK (TERMS_ACCEPTED IN ('Y', 'N')),
    ACCOUNT_STATUS      VARCHAR2(20) DEFAULT 'PENDING' CHECK (ACCOUNT_STATUS IN ('PENDING', 'ACTIVE', 'BLOCKED')),
    CREATED_DATE        DATE DEFAULT SYSDATE,
    UPDATED_DATE        DATE,
    CREATED_IP          VARCHAR2(45),
    LAST_LOGIN_DATE     DATE,
    LAST_LOGIN_IP       VARCHAR2(45)
);


ALTER TABLE TEMP_USER_MASTER ADD (
    USERNAME VARCHAR2(100),

    RESET_TOKEN_EXPIRY DATE,
    USER_ROLE VARCHAR2(50) DEFAULT 'USER',
