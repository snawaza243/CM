CREATE TABLE Clients (
    ClientId NUMBER PRIMARY KEY,
    CompanyName VARCHAR2(100) NOT NULL,
    ContactFirstName VARCHAR2(50) NOT NULL,
    ContactLastName VARCHAR2(50) NOT NULL,
    Email VARCHAR2(255),
    Phone VARCHAR2(20),
    Address VARCHAR2(255),
    City VARCHAR2(50),
    State VARCHAR2(50),
    ZipCode VARCHAR2(20),
    CreatedDate DATE DEFAULT SYSDATE NOT NULL,
    IsActive NUMBER(1) DEFAULT 1 NOT NULL
);

CREATE SEQUENCE seq_clients START WITH 1 INCREMENT BY 1;
CREATE OR REPLACE TRIGGER trg_clients_id
BEFORE INSERT ON Clients
FOR EACH ROW
BEGIN
    IF :NEW.ClientId IS NULL THEN
        SELECT seq_clients.NEXTVAL INTO :NEW.ClientId FROM dual;
    END IF;
END;
/




CREATE TABLE Employees (
    EmployeeId NUMBER PRIMARY KEY,

CREATE TABLE Projects (
