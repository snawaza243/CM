CREATE TABLE Clients (
    ClientId INT PRIMARY KEY,
    CompanyName VARCHAR2(100) NOT NULL,
    ContactFirstName VARCHAR2(50) NOT NULL,
    ContactLastName VARCHAR2(50) NOT NULL,
    Email VARCHAR2(255),
    Phone VARCHAR2(20),
    Address VARCHAR2(255),
    City VARCHAR2(50),
    State VARCHAR2(50),
    ZipCode VARCHAR2(20),
    CreatedDate DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    IsActive BIT NOT NULL DEFAULT 1
);

CREATE TABLE Employees (
    EmployeeId INT PRIMARY KEY IDENTITY(1,1),
    UserId VARCHAR2(450) NULL, -- Foreign Key to ASP.NET Identity 'AspNetUsers' table
    FirstName VARCHAR2(50) NOT NULL,
    LastName VARCHAR2(50) NOT NULL,
    Email VARCHAR2(255),
    Phone VARCHAR2(20),
    HireDate DATE NOT NULL,
    Position VARCHAR2(50), -- e.g., Project Manager, Foreman, Laborer
    HourlyRate DECIMAL(19,4) NULL, -- NULL if salaried
    Salary DECIMAL(19,4) NULL, -- NULL if hourly
    IsActive BIT NOT NULL DEFAULT 1
);

CREATE TABLE Projects (
    ProjectId INT PRIMARY KEY IDENTITY(1,1),
    ClientId INT NOT NULL FOREIGN KEY REFERENCES Clients(ClientId),
    ProjectName VARCHAR2(100) NOT NULL,
    Description VARCHAR2(MAX),
    StartDate DATE,
    EstimatedEndDate DATE,
    ActualEndDate DATE NULL,
    ContractValue DECIMAL(19,4) NOT NULL,
    Status VARCHAR2(20) NOT NULL DEFAULT 'Pending', -- e.g., Pending, Active, On-Hold, Completed, Cancelled
    Address VARCHAR2(255),
    CreatedDate DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    CreatedBy INT NOT NULL FOREIGN KEY REFERENCES Employees(EmployeeId) -- Who created this project in the system
);

CREATE TABLE ProjectPhases (
    PhaseId INT PRIMARY KEY IDENTITY(1,1),
    ProjectId INT NOT NULL FOREIGN KEY REFERENCES Projects(ProjectId) ON DELETE CASCADE,
    PhaseName VARCHAR2(50) NOT NULL, -- e.g., "Initial Excavation"
    Description VARCHAR2(255),
    EstimatedStartDate DATE,
    EstimatedEndDate DATE,
    ActualStartDate DATE NULL,
    ActualEndDate DATE NULL,
    Budget DECIMAL(19,4),
    SortOrder INT -- To order the phases correctly
);

CREATE TABLE Tasks (
    TaskId INT PRIMARY KEY IDENTITY(1,1),
    PhaseId INT NOT NULL FOREIGN KEY REFERENCES ProjectPhases(PhaseId) ON DELETE CASCADE,
    TaskName VARCHAR2(100) NOT NULL,
    Description VARCHAR2(MAX),
    AssignedToEmployeeId INT NULL FOREIGN KEY REFERENCES Employees(EmployeeId), -- Who is responsible
    DueDate DATE,
    Status VARCHAR2(20) NOT NULL DEFAULT 'NotStarted', -- e.g., NotStarted, InProgress, Completed, Blocked
    Priority VARCHAR2(20) NOT NULL DEFAULT 'Medium', -- e.g., Low, Medium, High, Critical
    CreatedDate DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

CREATE TABLE Invoices (
    InvoiceId INT PRIMARY KEY IDENTITY(1,1),
    ProjectId INT NOT NULL FOREIGN KEY REFERENCES Projects(ProjectId),
    InvoiceNumber VARCHAR2(50) NOT NULL UNIQUE, -- e.g., INV-2023-001
    InvoiceDate DATE NOT NULL DEFAULT GETUTCDATE(),
    DueDate DATE NOT NULL,
    Status VARCHAR2(20) NOT NULL DEFAULT 'Draft', -- e.g., Draft, Sent, Viewed, Paid, Overdue, Cancelled
    TotalAmount DECIMAL(19,4) NOT NULL,
    TaxAmount DECIMAL(19,4) DEFAULT 0,
    Notes VARCHAR2(MAX),
    CreatedDate DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);


CREATE TABLE InvoiceLineItems (
    LineItemId INT PRIMARY KEY IDENTITY(1,1),
    InvoiceId INT NOT NULL FOREIGN KEY REFERENCES Invoices(InvoiceId) ON DELETE CASCADE,
    Description VARCHAR2(255) NOT NULL, -- e.g., "Phase 1: Foundation Work"
    Quantity DECIMAL(19,4) NOT NULL DEFAULT 1,
    UnitPrice DECIMAL(19,4) NOT NULL,
    LineTotal AS (Quantity * UnitPrice) -- Computed Column
);


CREATE TABLE Payments (
    PaymentId INT PRIMARY KEY IDENTITY(1,1),
    InvoiceId INT NOT NULL FOREIGN KEY REFERENCES Invoices(InvoiceId),
    Amount DECIMAL(19,4) NOT NULL,
    PaymentDate DATE NOT NULL DEFAULT GETUTCDATE(),
    PaymentMethod VARCHAR2(50) NOT NULL, -- e.g., Check, Bank Transfer, Credit Card, Cash
    ReferenceNumber VARCHAR2(100), -- Check number or transaction ID
    Notes VARCHAR2(255),
    RecordedBy INT NOT NULL FOREIGN KEY REFERENCES Employees(EmployeeId), -- Who entered this payment
    RecordedDate DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

CREATE TABLE Expenses (
    ExpenseId INT PRIMARY KEY IDENTITY(1,1),
    ProjectId INT NOT NULL FOREIGN KEY REFERENCES Projects(ProjectId),
    Category VARCHAR2(50) NOT NULL, -- e.g., Materials, Labor, Equipment, Subcontractor
    Description VARCHAR2(255) NOT NULL,
    Amount DECIMAL(19,4) NOT NULL,
    ExpenseDate DATE NOT NULL,
    Supplier VARCHAR2(100),
    ReceiptImageUrl VARCHAR2(MAX), -- Path to the uploaded receipt image
    RecordedBy INT NOT NULL FOREIGN KEY REFERENCES Employees(EmployeeId),
    RecordedDate DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

CREATE TABLE TimeLogs (
    TimeLogId INT PRIMARY KEY IDENTITY(1,1),
    EmployeeId INT NOT NULL FOREIGN KEY REFERENCES Employees(EmployeeId),
    TaskId INT NOT NULL FOREIGN KEY REFERENCES Tasks(TaskId),
    ProjectId INT NOT NULL FOREIGN KEY REFERENCES Projects(ProjectId), -- Denormalized for easier querying
    LogDate DATE NOT NULL,
    HoursWorked DECIMAL(5,2) NOT NULL, -- e.g., 7.5 hours
    Description VARCHAR2(255), -- What was done during these hours?
    IsApproved BIT NOT NULL DEFAULT 0, -- Needs manager approval for payroll
    ApprovedBy INT NULL FOREIGN KEY REFERENCES Employees(EmployeeId),
    ApprovedDate DATETIME2 NULL
);
