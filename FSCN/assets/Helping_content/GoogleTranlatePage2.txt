            pageLanguage: 'en',

            layout: google.translate.TranslateElement.InlineLayout.SIMPLE
        }, 'google_translate_element');

        <script>
            $(document).ready(function () {

                // Toggle dropdown
                $('#langDropdown').click(function (e) {
                    e.preventDefault();
                    $('#langMenu').toggle();
                });

                // Click outside closes dropdown
                $(document).click(function (e) {
                    if (!$(e.target).closest('#langDropdown').length && !$(e.target).closest('#langMenu').length) {
                        $('#langMenu').hide();
                    }
                });

                // Set Google Translate language
                function setGoogleLang(lang) {
                    var checkExist = setInterval(function () {
                        var $sel = $('.goog-te-combo');
                        if ($sel.length) {
                            clearInterval(checkExist);
                            $sel.val(lang).trigger('change');
                            var sel = $sel.get(0);
                            if (sel) {
                                var ev = document.createEvent('HTMLEvents');
                                ev.initEvent('change', true, true);
                                sel.dispatchEvent(ev);
                            }
                        }
                    }, 100);
                }

                // Language selection
                $('.lang-select').click(function (e) {
                    e.preventDefault();
                    var lang = $(this).data('lang');
                    setGoogleLang(lang);
                    $('#langMenu').hide();
                });

            });
        </script>
