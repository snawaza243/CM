CREATE OR REPLACE PROCEDURE PROC_TEMP_USER_INSERT (
    P_FIRST_NAME        IN VARCHAR2,
    P_LAST_NAME         IN VARCHAR2,
    P_EMAIL             IN VARCHAR2,
    P_PHONE             IN VARCHAR2,
    P_PASSWORD_HASH     IN VARCHAR2,
    P_MARKETING_EMAILS  IN CHAR,
    P_TERMS_ACCEPTED    IN CHAR,
    P_CREATED_IP        IN VARCHAR2,
    P_USERNAME          IN VARCHAR2,
    P_USER_ROLE         IN VARCHAR2,
    P_SOURCE            IN VARCHAR2,
    P_STATUS_CODE       OUT VARCHAR2,   -- 'SUCCESS' or 'ERROR'
    P_STATUS_MESSAGE    OUT VARCHAR2    -- detailed message
)
AS
    -- Local variables for error handling
    v_error_code   NUMBER;
    v_error_msg    VARCHAR2(4000);
      v_count NUMBER;
BEGIN
    -- Create a savepoint before insertion (for rollback safety)
    SAVEPOINT before_insert;

      

    -- 1️⃣ Validate email uniqueness
    SELECT COUNT(*) INTO v_count 
    FROM TEMP_USER_MASTER 
    WHERE EMAIL = P_EMAIL;

    IF v_count > 0 THEN
        P_STATUS_CODE := 'ERROR';
        P_STATUS_MESSAGE := 'Email already exists. Please use a different email.';
        RETURN;
    END IF;

    -- 2️⃣ Validate username uniqueness
    SELECT COUNT(*) INTO v_count
    FROM TEMP_USER_MASTER
    WHERE USERNAME = P_USERNAME;

    IF v_count > 0 THEN
        P_STATUS_CODE := 'ERROR';
        P_STATUS_MESSAGE := 'Username already exists. Please choose another.';
        RETURN;
    END IF;

    -- 3️⃣ Insert user if validation passes
    INSERT INTO temp_user_master (
        user_id,
        first_name,
        last_name,
        email,
        phone,
        password_hash,
        marketing_emails,
        terms_accepted,
        account_status,
        created_date,
        updated_date,
        created_ip,
        username,
        email_verified,
        user_role,
        source
    ) VALUES (
        seq_temp_user_master.NEXTVAL,
        p_first_name,
        p_last_name,
        p_email,
        p_phone,
        p_password_hash,
        p_marketing_emails,
        p_terms_accepted,
        'ACTIVE',
        sysdate,
        sysdate,
        p_created_ip,
        p_username,
        'N',
        p_user_role,
        p_source
    ) ;

    -- ✅ Success
    P_STATUS_CODE := 'SUCCESS';
    P_STATUS_MESSAGE := 'User created successfully.';
    -- Commit the transaction if all is good
    COMMIT; EXCEPTION
    WHEN OTHERS THEN
        -- Capture error details
        v_error_code := sqlcode;
        v_error_msg := sqlerrm;
        p_status_code := 'ERROR';
        p_status_message := v_error_msg;

        -- Rollback to before the insert
        ROLLBACK TO before_insert;

        -- Optional: log error to an error table for audit
        INSERT INTO error_log_table (
            error_code,
            error_message,
            log_date,
            module_name
        ) VALUES (
            v_error_code,
            v_error_msg,
            sysdate,
            'TEMP_USER_INSERT'
        );

        -- Commit the log entry
        COMMIT;

        -- Re-raise the error to propagate to application
        RAISE;
END proc_temp_user_insert;
/
