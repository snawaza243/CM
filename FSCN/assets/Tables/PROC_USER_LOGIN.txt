CREATE OR REPLACE NONEDITIONABLE PROCEDURE PROC_USER_LOGIN(
    P_EMAIL        IN VARCHAR2,
    P_PASSWORD     IN VARCHAR2,
    P_USER_ROLE    IN VARCHAR2,
    P_LOGIN_IP     IN VARCHAR2,
    P_STATUS_CODE  OUT VARCHAR2,   -- 'SUCCESS' or 'ERROR'
    P_STATUS_MESSAGE OUT VARCHAR2  -- detailed message
)
AS
    v_count NUMBER;
    v_hash_pass varchar2(100);
BEGIN
    -- Check credentials
    SELECT COUNT(*)
    INTO v_count
    FROM TEMP_USER_MASTER
    WHERE EMAIL = P_EMAIL
      AND PASSWORD_HASH = P_PASSWORD
      AND USER_ROLE = P_USER_ROLE
      AND ACCOUNT_STATUS = 'ACTIVE';

    IF v_count = 1 THEN
        -- Update last login date and IP
        UPDATE TEMP_USER_MASTER
        SET LAST_LOGIN_DATE = SYSDATE,
            LAST_LOGIN_IP   = P_LOGIN_IP
        WHERE EMAIL = P_EMAIL
          AND USER_ROLE = P_USER_ROLE;

        P_STATUS_CODE := 'SUCCESS';
        P_STATUS_MESSAGE := 'Login successful!';
    ELSE
        P_STATUS_CODE := 'ERROR';
        
        select PASSWORD_HASH into v_hash_pass from TEMP_USER_MASTER WHERE EMAIL = P_EMAIL
          AND USER_ROLE = P_USER_ROLE;
        P_STATUS_MESSAGE := v_hash_pass;
    END IF;

EXCEPTION
    WHEN OTHERS THEN
        P_STATUS_CODE := 'ERROR';
        P_STATUS_MESSAGE := 'Error: ' || SQLERRM;
END PROC_USER_LOGIN;
/
